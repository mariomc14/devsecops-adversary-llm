# SCE 1.8 Experiment: Encrypted CodeBuild Artifact Control

## Analysis of Previous Execution Issues

Based on the execution log and quality reports, the critical failure was:

1. **CloudFormation Stack Creation Failed** → Stack rolled back before CodeBuild project could be created
2. **Attack Could Not Execute** → ResourceNotFoundException on CodeBuild project start-build call
3. **Probe Could Not Verify** → No resources existed to validate

**Root Cause**: The CloudFormation template had an error that caused `ROLLBACK_IN_PROGRESS`. This prevented the CodeBuild project from being created, which then caused both the attack and probe to fail.

**Key Improvements for This Implementation**:
- Validate CloudFormation template syntax before deployment
- Add detailed error logging for stack creation failures
- Implement proper retry logic with exponential backoff
- Add validation that resources actually exist before attempting attack
- Enhance proof-of-concept evidence collection

def retry_with_backoff(func, max_retries: int = MAX_RETRIES, description: str = ""):
    """
    Execute function with exponential backoff retry logic.
    
    Args:
        func: Callable to execute
        max_retries: Maximum number of retry attempts
        description: Description of the operation for logging
    
    Returns:
        Result of func() on success
        
    Raises:
        Exception if all retries exhausted
    """
    start_time = time.monotonic()
    last_error = None
    
    for attempt in range(max_retries):
        try:
            log_debug(f"Attempt {attempt + 1}/{max_retries}: {description}")
            result = func()
            elapsed = time.monotonic() - start_time
            log_debug(f"Success after {elapsed:.2f}s")
            return result
        except Exception as e:
            last_error = e
            if attempt < max_retries - 1:
                backoff_seconds = RETRY_BACKOFF_BASE ** attempt
                log_warning(f"Attempt {attempt + 1} failed: {str(e)[:100]}. "
                           f"Retrying in {backoff_seconds}s...")
                time.sleep(backoff_seconds)
            else:
                log_error(f"All {max_retries} attempts exhausted for: {description}")
    
    raise last_error

def hypothesis_verification() -> bool:
    """
    Phase 3: Verify preventive control is working (Probe).
    
    Validates that:
    1. S3 bucket has encryption enabled (AES256)
    2. S3 bucket policy denies unencrypted uploads (PREVENTIVE)
    3. CodeBuild project has encryption enforced
    4. No unencrypted objects exist in the bucket
    5. Build attempt was blocked or credentials were not exfiltrated
    
    Returns:
        True if preventive control is effective, False otherwise
    """
    log_info("=== Verifying preventive control ===")
    
    if not experiment_state["resources_created"]:
        log_error("Resources not created - cannot verify hypothesis")
        return False
    
    s3 = boto3.client("s3", region_name=REGION)
    codebuild = boto3.client("codebuild", region_name=REGION)
    
    bucket_name = experiment_state["bucket_name"]
    project_name = experiment_state["codebuild_project"]
    
    try:
        log_info("Performing preventive control verification checks...")
        all_checks_passed = True
        
        # Check 1: Verify S3 bucket encryption is enabled
        log_info("\n[Check 1/5] Verifying S3 bucket encryption (AES256)...")
        try:
            encryption = s3.get_bucket_encryption(Bucket=bucket_name)
            rules = encryption['ServerSideEncryptionConfiguration']['Rules']
            
            encryption_enabled = False
            for rule in rules:
                sse = rule['ApplyServerSideEncryptionByDefault']
                if sse.get('SSEAlgorithm') == 'AES256':
                    encryption_enabled = True
                    log_info("  ✓ S3 bucket encryption enabled with AES256")
                    break
            
            if not encryption_enabled:
                log_error("  ✗ S3 bucket encryption not properly configured")
                all_checks_passed = False
                
        except ClientError as e:
            if e.response['Error']['Code'] == 'ServerSideEncryptionConfigurationNotFoundError':
                log_error("  ✗ S3 bucket encryption not configured")
                all_checks_passed = False
            else:
                raise
        
        # Check 2: Verify S3 bucket policy denies unencrypted uploads
        log_info("[Check 2/5] Verifying S3 bucket policy denies unencrypted uploads...")
        try:
            policy_response = s3.get_bucket_policy(Bucket=bucket_name)
            policy = json.loads(policy_response['Policy'])
            
            deny_unencrypted_found = False
            for statement in policy.get('Statement', []):
                if (statement.get('Effect') == 'Deny' and
                    'PutObject' in statement.get('Action', []) and
                    'Condition' in statement):
                    
                    conditions = statement['Condition']
                    if 'StringNotEquals' in conditions:
                        if 's3:x-amz-server-side-encryption' in conditions['StringNotEquals']:
                            deny_unencrypted_found = True
                            log_info("  ✓ Bucket policy enforces encrypted uploads (preventive)")
                            break
            
            if not deny_unencrypted_found:
                log_error("  ✗ Bucket policy does not deny unencrypted uploads")
                all_checks_passed = False
                
        except ClientError as e:
            if e.response['Error']['Code'] == 'NoSuchBucketPolicy':
                log_error("  ✗ No bucket policy found")
                all_checks_passed = False
            else:
                raise
        
        # Check 3: Verify CodeBuild project encryption is enabled
        log_info("[Check 3/5] Verifying CodeBuild project encryption enforcement...")
        try:
            project_response = codebuild.batch_get_projects(names=[project_name])
            projects = project_response.get('projects', [])
            
            if not projects:
                log_error("  ✗ CodeBuild project not found")
                all_checks_passed = False
            else:
                project = projects[0]
                artifacts = project.get('artifacts', {})
                encryption_disabled = artifacts.get('encryptionDisabled', False)
                
                if not encryption_disabled:
                    log_info("  ✓ CodeBuild project encryption enabled")
                else:
                    log_error("  ✗ CodeBuild project encryption is disabled")
                    all_checks_passed = False
                    
        except ClientError as e:
            log_error(f"  ✗ Error checking CodeBuild project: {e}")
            all_checks_passed = False
        
        # Check 4: Verify S3 bucket public access blocking
        log_info("[Check 4/5] Verifying S3 public access blocking configuration...")
        try:
            pab = s3.get_public_access_block(Bucket=bucket_name)
            config = pab['PublicAccessBlockConfiguration']
            
            all_blocked = all([
                config.get('BlockPublicAcls', False),
                config.get('BlockPublicPolicy', False),
                config.get('IgnorePublicAcls', False),
                config.get('RestrictPublicBuckets', False)
            ])
            
            if all_blocked:
                log_info("  ✓ All public access blocks enabled")
            else:
                log_error("  ✗ Some public access blocks not enabled")
                all_checks_passed = False
                
        except ClientError as e:
            log_error(f"  ✗ Error checking public access block: {e}")
            all_checks_passed = False
        
        # Check 5: Verify no unencrypted objects in bucket
        log_info("[Check 5/5] Scanning bucket for unencrypted objects...")
        try:
            unencrypted_objects = []
            paginator = s3.get_paginator('list_objects_v2')
            pages = paginator.paginate(Bucket=bucket_name)
            
            for page in pages:
                if 'Contents' not in page:
                    continue
                
                for obj in page['Contents']:
                    key = obj['Key']
                    
                    # Get object metadata to check encryption
                    try:
                        head_response = s3.head_object(Bucket=bucket_name, Key=key)
                        
                        # Check if object has server-side encryption
                        server_side_encryption = head_response.get('ServerSideEncryption')
                        
                        if not server_side_encryption:
                            unencrypted_objects.append(key)
                            log_warning(f"    Found unencrypted object: {key}")
                        else:
                            log_debug(f"    Object {key} encrypted with {server_side_encryption}")
                            
                    except ClientError as e:
                        log_warning(f"    Could not verify encryption for {key}: {e}")
