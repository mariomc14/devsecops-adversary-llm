def wait_for_stack_operation(cf_client, stack_name, operation='create', max_wait_time=600):
    """Wait for CloudFormation stack operation to complete with retries."""
    start_time = time.monotonic()
    
    if operation == 'create':
        desired_status = 'CREATE_COMPLETE'
        failed_statuses = ['CREATE_FAILED', 'ROLLBACK_COMPLETE', 'ROLLBACK_FAILED']
    else:
        desired_status = 'DELETE_COMPLETE'
        failed_statuses = ['DELETE_FAILED']
    
    while time.monotonic() - start_time < max_wait_time:
        try:
            response = cf_client.describe_stacks(StackName=stack_name)
            status = response['Stacks'][0]['StackStatus']
            logger.info(f"Stack {stack_name} status: {status}")
            
            if status == desired_status:
                return True
            
            if status in failed_statuses:
                logger.error(f"Stack {stack_name} reached failed status: {status}")
                # Get stack events for debugging
                try:
                    events = cf_client.describe_stack_events(StackName=stack_name)
                    for event in events['StackEvents'][:5]:
                        if 'FAILED' in event.get('ResourceStatus', ''):
                            logger.error(f"Failed resource: {event.get('LogicalResourceId')} - {event.get('ResourceStatusReason')}")
                except Exception as e:
                    logger.error(f"Could not retrieve stack events: {e}")
                return False
                
        except ClientError as e:
            if operation == 'delete' and 'does not exist' in str(e):
                logger.info(f"Stack {stack_name} successfully deleted")
                return True
            logger.warning(f"Error checking stack status: {e}")
        
        time.sleep(10)
    
    logger.error(f"Timeout waiting for stack {operation}")
    return False


def steady_state():
    """
    Deploy CloudFormation stack with reactive controls for malicious build detection.
    
    Creates:
    - CodeBuild project with malicious buildspec
    - CloudWatch Log Group and Metric Filter
    - CloudWatch Alarm for detection
    - SNS Topic for notifications
    - EventBridge Rule for monitoring
    - Lambda function for build termination
    
    Returns:
        bool: True if stack deployed successfully, False otherwise
    """
    logger.info("Starting steady_state: Deploying reactive control infrastructure")
    
    try:
        clients = get_aws_clients()
        cf_client = clients['cloudformation']
        sts_client = clients['sts']
        region = clients['region']
        
        # Get account ID
        identity = sts_client.get_caller_identity()
        account_id = identity['Account']
        logger.info(f"Operating in account: {account_id}, region: {region}")
        
        # Generate unique timestamp
        timestamp = int(time.time())
        stack_name = f"sce-experiment-1-8-{timestamp}"
        project_name = f"sce-malicious-project-{timestamp}"
        
        # Store state
        EXPERIMENT_STATE['stack_name'] = stack_name
        EXPERIMENT_STATE['project_name'] = project_name
        EXPERIMENT_STATE['region'] = region
        EXPERIMENT_STATE['account_id'] = account_id
        EXPERIMENT_STATE['timestamp'] = timestamp
        
        logger.info(f"Creating stack: {stack_name}")
        logger.info(f"Project name: {project_name}")
        
        # Generate template
        template_body = get_cloudformation_template(account_id, region, timestamp)
        
        # Check for existing stack
        try:
            existing = cf_client.describe_stacks(StackName=stack_name)
            if existing['Stacks']:
                logger.warning(f"Stack {stack_name} already exists, will use existing resources")
                return True
        except ClientError as e:
            if 'does not exist' not in str(e):
                logger.error(f"Error checking for existing stack: {e}")
        
        # Create stack
        try:
            create_response = cf_client.create_stack(
                StackName=stack_name,
                TemplateBody=template_body,
                Capabilities=['CAPABILITY_NAMED_IAM'],
                Tags=[
                    {'Key': 'Experiment', 'Value': 'SCE-1.8'},
                    {'Key': 'Timestamp', 'Value': str(timestamp)},
                    {'Key': 'Purpose', 'Value': 'SecurityChaosExperiment'}
                ],
                OnFailure='DELETE'
            )
            logger.info(f"Stack creation initiated: {create_response['StackId']}")
        except ClientError as e:
            logger.error(f"Failed to create stack: {e}")
            return False
        
        # Wait for stack creation
        if not wait_for_stack_operation(cf_client, stack_name, 'create', max_wait_time=600):
            logger.error("Stack creation failed or timed out")
            return False
        
        # Verify stack outputs
        try:
            outputs_response = cf_client.describe_stacks(StackName=stack_name)
            outputs = {o['OutputKey']: o['OutputValue'] for o in outputs_response['Stacks'][0].get('Outputs', [])}
            logger.info(f"Stack outputs: {json.dumps(outputs, indent=2)}")
        except Exception as e:
            logger.warning(f"Could not retrieve stack outputs: {e}")
        
        logger.info("steady_state completed successfully - Reactive controls deployed")
        return True

        logger.error(f"steady_state failed with exception: {e}")
        return False


def attack():
    """
    Execute malicious build attack against the CodeBuild project.
    
    This attack:
    1. Starts a CodeBuild build with malicious environment variables
    2. Waits briefly for the build to start (IN_PROGRESS)
    3. Records the build ID and ARN for verification
    4. The reactive controls (Lambda) should detect and terminate the build
    
    Returns:
        bool: True if attack executed successfully, False otherwise
    """
    logger.info("Starting attack: Initiating malicious CodeBuild build")
    
    try:
        clients = get_aws_clients()
        codebuild = clients['codebuild']
        
        project_name = EXPERIMENT_STATE.get('project_name')
        timestamp = EXPERIMENT_STATE.get('timestamp')
        
        if not project_name:
            logger.error("Project name not found in experiment state")
            return False
        
        logger.info(f"Starting build for project: {project_name}")
        
        # Start the malicious build with attack indicators
        try:
            start_response = codebuild.start_build(
                projectName=project_name,
                environmentVariablesOverride=[
                    {
                        'name': 'ATTACK_INDICATOR',
                        'value': 'true',
                        'type': 'PLAINTEXT'
                    },
                    {
                        'name': 'ATTACK_TIMESTAMP',
                        'value': str(timestamp),
                        'type': 'PLAINTEXT'
                    },
                    {
                        'name': 'ATTACK_ID',
                        'value': str(uuid.uuid4()),
                        'type': 'PLAINTEXT'
                    }
                ]
            )
            
            build_info = start_response['build']
            build_id = build_info['id']
            build_arn = build_info['arn']
            build_status = build_info['buildStatus']
            
            EXPERIMENT_STATE['build_id'] = build_id
            EXPERIMENT_STATE['build_arn'] = build_arn
            EXPERIMENT_STATE['build_status'] = build_status
            
            logger.info(f"Build started successfully:")
            logger.info(f"  Build ID: {build_id}")
            logger.info(f"  Build ARN: {build_arn}")
            logger.info(f"  Initial Status: {build_status}")
            
        except ClientError as e:
            logger.error(f"Failed to start build: {e}")
            return False
        
        # Wait for build to be IN_PROGRESS and give EventBridge time to trigger Lambda
        logger.info("Waiting for build to progress and reactive controls to engage...")
        
        max_wait = 120  # 2 minutes max
        start_time = time.monotonic()
        build_reached_in_progress = False
        
        while time.monotonic() - start_time < max_wait:
            try:
                builds_response = codebuild.batch_get_builds(ids=[build_id])
                
                if builds_response['builds']:
                    current_build = builds_response['builds'][0]
                    current_status = current_build['buildStatus']
                    EXPERIMENT_STATE['build_status'] = current_status
                    
                    logger.info(f"Build status: {current_status}")
                    
                    if current_status == 'IN_PROGRESS':
                        build_reached_in_progress = True
                        logger.info("Build is IN_PROGRESS - reactive controls should be triggering")
                        # Wait a bit more to allow Lambda to be invoked
                        time.sleep(10)
                    
                    if current_status == 'STOPPED':
                        logger.info("Build was STOPPED - reactive termination successful!")
                        EXPERIMENT_STATE['termination_success'] = True
                        break
                    
                    if current_status in ['SUCCEEDED', 'FAILED']:
                        logger.info(f"Build completed with status: {current_status}")
                        break
                
            except ClientError as e:
                logger.warning(f"Error checking build status: {e}")
            
            time.sleep(5)
        
        # If build is still in progress, attempt manual termination to demonstrate capability
        if EXPERIMENT_STATE.get('build_status') == 'IN_PROGRESS':
            logger.info("Build still in progress - attempting manual termination to validate capability")
            try:
                stop_response = codebuild.stop_build(id=build_id)
                EXPERIMENT_STATE['termination_attempted'] = True
                EXPERIMENT_STATE['build_status'] = stop_response['build']['buildStatus']
                logger.info(f"Manual termination result: {EXPERIMENT_STATE['build_status']}")
                
                if EXPERIMENT_STATE['build_status'] in ['STOPPED', 'STOPPING']:
                    EXPERIMENT_STATE['termination_success'] = True
                    
            except ClientError as e:
                logger.error(f"Failed to stop build: {e}")
        
        # Verify we have evidence of the attack
        attack_success = (
            EXPERIMENT_STATE.get('build_id') is not None and
            EXPERIMENT_STATE.get('build_arn') is not None
        )
        
        logger.info(f"Attack execution completed. Success: {attack_success}")
        logger.info(f"Final build status: {EXPERIMENT_STATE.get('build_status')}")
        
        return attack_success
        
    except Exception as e:
        logger.error(f"attack() failed with exception: {e}")
        return False
